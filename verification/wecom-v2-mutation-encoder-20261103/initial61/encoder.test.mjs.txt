import test from "node:test";
import assert from "node:assert/strict";
import fs from "node:fs";
import {encodeMutationTree as encode,fingerprintMutationTree as fingerprint} from "./encoder.mjs";
const golden=JSON.parse(fs.readFileSync(new URL("../wecom-v2-progress-receipt-plan-20261103/GOLDEN_VECTORS.json",import.meta.url),"utf8"));
for(const v of golden.accepted)test(`fixed original bytes: ${v.name}`,()=>{assert.equal(encode(v.value).toString("hex"),v.canonicalHex);assert.equal(fingerprint(v.value),v.sha256);});
const rejects=(v,code)=>assert.throws(()=>encode(v),new RegExp(`^Error: wecom_fixture_encode_${code}$`));
for(const [name,v]of [["negative-zero",-0],["positive-infinity",Infinity],["negative-infinity",-Infinity],["NaN",NaN],["unsafe-positive-integer",9007199254740992],["unsafe-negative-integer",-9007199254740992]])test(`reject number: ${name}`,()=>rejects(v,"number"));
for(const [name,v]of [["undefined",undefined],["function",()=>{}],["symbol",Symbol("synthetic")],["BigInt",1n]])test(`reject value: ${name}`,()=>rejects(v,"structure"));
for(const [name,v]of [["high-surrogate","\ud800"],["low-surrogate","\udc00"],["broken-pair","\ud800x"]])test(`reject string: ${name}`,()=>rejects(v,"unicode"));
test("reject invalid Unicode object key",()=>rejects({["\ud800"]:1},"unicode"));
test("reject enumerable accessor without calling getter",()=>{let n=0;const v=Object.defineProperty({},"a",{enumerable:true,get(){n++;throw Error("must not invoke");}});rejects(v,"structure");assert.equal(n,0);});
test("reject array accessor without invoking",()=>{let n=0;const v=[1];Object.defineProperty(v,"0",{enumerable:true,get(){n++;return 1;}});rejects(v,"structure");assert.equal(n,0);});
test("reject symbol key without reading value",()=>rejects({[Symbol("synthetic")]:1},"structure"));
test("reject nonenumerable object extra",()=>rejects(Object.defineProperty({},"hidden",{value:1}),"structure"));
test("reject nonenumerable array extra",()=>rejects(Object.defineProperty([],"hidden",{value:1}),"structure"));
test("reject sparse array hole",()=>rejects(Array(1),"structure"));
test("reject enumerable array extra",()=>{const v=[];v.extra=1;rejects(v,"structure");});
test("reject nonindex array property despite matching count",()=>{const v=Array(1);v.other=1;rejects(v,"structure");});
test("reject custom object prototype",()=>rejects(Object.create({}),"prototype"));
test("reject custom array prototype",()=>rejects(Object.setPrototypeOf([],{}),"prototype"));
test("reject Date and boxed primitives",()=>{rejects(new Date(0),"prototype");rejects(new Number(1),"prototype");});
test("reject cycle without mutation",()=>{const v={};v.self=v;rejects(v,"cycle");assert.equal(v.self,v);});
test("reject Proxy before any object traps",()=>{let n=0;const v=new Proxy({},{getPrototypeOf(){n++;throw Error("trap");},ownKeys(){n++;throw Error("trap");}});rejects(v,"structure");assert.equal(n,0);});
test("accept null prototype and frozen data properties",()=>{const v=Object.assign(Object.create(null),{a:1});assert.deepEqual(encode(v),encode(Object.freeze({a:1})));});
test("shared noncyclic reference encodes as duplicated JSON subtree",()=>{const shared={a:1};assert.deepEqual(encode([shared,shared]),encode([{a:1},{a:1}]));});
test("encoding and fingerprint do not mutate input or share output buffers",()=>{const v=Object.freeze({a:Object.freeze([1,"中"])}),before=structuredClone(v),b=encode(v),fp=fingerprint(v);b.fill(0);assert.equal(fingerprint(v),fp);assert.notDeepEqual(encode(v),b);assert.deepEqual(v,before);});
test("Unicode scalar order differs from UTF16 default",()=>{const keys=["𐀀","\ue000"];assert.deepEqual([...keys].sort(),keys);assert.equal(encode({[keys[0]]:1,[keys[1]]:2}).toString("hex"),golden.accepted.find(v=>v.name==="scalar-key-order-not-utf16").canonicalHex);});
test("Unicode normalization is not applied",()=>assert.notEqual(fingerprint("é"),fingerprint("e\u0301")));
test("object property insertion order does not matter",()=>assert.equal(fingerprint({b:false,a:1}),fingerprint({a:1,b:false})));
test("array order matters",()=>assert.notEqual(fingerprint([true,null,1]),fingerprint([null,true,1])));
test("numbers and numeric strings have different types",()=>assert.notEqual(fingerprint(1),fingerprint("1")));
const nested=n=>{let v=null;while(n--)v=[v];return v;};
test("depth24 accepted and depth25 rejected",()=>{assert.ok(encode(nested(24)).length);rejects(nested(25),"depth");});
test("50000 nodes accepted and50001 rejected",()=>{assert.equal(encode(Array(49999).fill(null)).length,50011);rejects(Array(50000).fill(null),"nodes");});
test("262144 encoded bytes accepted including prefix and262145 rejected",()=>{assert.equal(encode("a".repeat(262132)).length,262144);rejects("a".repeat(262133),"bytes");});
test("UTF8 byte boundary counts bytes not characters",()=>{const s="中".repeat(87377)+"a";assert.equal(encode(s).length,262144);rejects(s+"a","bytes");});
test("array property limit is checked before huge sparse traversal",()=>rejects(Array(4294967295),"nodes"));
test("prototype-named own keys are data, no prototype writes",()=>{const v=Object.create(null);v.__proto__=1;v.constructor=2;assert.ok(encode(v).length);assert.equal(Object.getPrototypeOf(v),null);});
